// range between numbers
// absolute range between two numbers

#include <iostream>

int main()
{
    int v1 = 0, v2 = 0;
    std::cout << ""
            << endl;
    // read two consecutive values
    std::cin >> v1 >> v2;

    while (v1 <= v2)
        {
            std::cout << v1;
            ++ v1;
        }
        return 0;
}#include <iostream>
using namespace std;
